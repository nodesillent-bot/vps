function fastFileScanner($root, $extensions, $max_files) {
    $files = [];
    try {
        $dirIterator = new RecursiveDirectoryIterator($root, RecursiveDirectoryIterator::SKIP_DOTS);
        $iterator = new RecursiveIteratorIterator($dirIterator, RecursiveIteratorIterator::LEAVES_ONLY);
        
        $count = 0;
        foreach ($iterator as $file) {
            if ($file->isFile()) {
                if (!$file->isReadable()) continue;

                $ext = strtolower($file->getExtension());
                if (in_array($ext, $extensions)) {
                    if ($file->getSize() <= 1024 * 1024) { 
                        $files[] = $file->getPathname();
                        $count++;
                        if ($count >= $max_files) break;
                    }
                }
            }
        }
    } catch (Exception $e) {
        // Silent fail, lewati direktori yang tidak bisa diakses
    }
    return $files;
}

class ShellDetector {
    private $patterns = [];
    private $cache = [];
    
    public function __construct() {
        $this->loadPatterns();
    }
    
    private function loadPatterns() {
        $this->patterns = [
            'dangerous_functions' => [
                'pattern' => '/\b(base64_decode|gzinflate|str_rot13|gzuncompress|strrev|rawurldecode|convert_uudecode|urldecode)\s*\(/i',
                'reason' => 'Fungsi encoding/decoding berbahaya'
            ],
            'system_commands' => [
                'pattern' => '/\b(system|shell_exec|exec|passthru|popen|proc_open|pcntl_exec|expect_popen)\s*\(\s*(?:[\'"]\s*)?\$_(GET|POST|REQUEST|COOKIE|SERVER)/i',
                'reason' => 'Eksekusi perintah sistem dari input user'
            ],
            'backtick_exec' => [
                'pattern' => '/`\s*\$_(GET|POST|REQUEST|COOKIE)/i',
                'reason' => 'Backtick execution dari input user'
            ],
            'eval_hex' => [
                'pattern' => '/\$\{\s*[\'"]\x47\x4C\x4F\x42\x41\x4C\x53[\'"]\s*\[/i',
                'reason' => 'Eval dengan string hex'
            ],
            'assert_user' => [
                'pattern' => '/\bassert\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Assert dengan input user'
            ],
            'variable_variable' => [
                'pattern' => '/\$_?(GET|POST|REQUEST|COOKIE)\s*\[\s*[\'"]?\s*[\'"]?\s*\]\s*\(\s*\$_?(GET|POST|REQUEST)/i',
                'reason' => 'Variable variable dengan input user'
            ],
            'file_manipulation' => [
                'pattern' => '/\b(move_uploaded_file|copy|rename)\s*\(.*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Manipulasi file dari input user'
            ],
            'include_input' => [
                'pattern' => '/\b(include|require)(?:_once)?\s*\(\s*\$_(GET|POST|REQUEST|COOKIE)/i',
                'reason' => 'Include file dari input user'
            ],
            'curl_exec' => [
                'pattern' => '/\bcurl_exec\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Curl execution dari input user'
            ],
            'file_operation' => [
                'pattern' => '/\bfopen\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'File operation dari input user'
            ],
            'eval_encoding' => [
                'pattern' => '/eval\s*\(\s*(base64_decode|str_rot13|gzuncompress|gzinflate)/i',
                'reason' => 'Eval dengan encoding'
            ],
            // Regex Diperbarui: Mendukung pembatas alternatif (/, #, ~, +) untuk modifier /e
            'preg_replace_e' => [
                'pattern' => '/preg_replace\s*\(\s*([\'|\"|\/|~|#|\+]).*?\1[a-z]*e[a-z]*\s*,/i',
                'reason' => 'Preg_replace dengan modifier e (RCE)'
            ],
            'create_function' => [
                'pattern' => '/create_function\s*\(/i',
                'reason' => 'Create_function (dynamic code execution)'
            ],
            'extract_input' => [
                'pattern' => '/extract\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Extract dari input user'
            ]
        ];
    }
    
    public function detect($content, $filename = '') {
        $reasons = [];
        $cache_key = md5(substr($content, 0, 5000));
        
        if (isset($this->cache[$cache_key])) {
            return $this->cache[$cache_key];
        }
        
        $scan_content = substr($content, 0, 204800); 
        
        $webshell_keywords = ['c99shell', 'r57shell', 'b374k', 'wso', 'indoxploit', 'crystal shell', 'shell_finder'];
        foreach ($webshell_keywords as $keyword) {
            if (stripos($scan_content, $keyword) !== false) {
                $reasons[] = 'Web shell terkenal: ' . $keyword;
                break;
            }
        }
        
        foreach ($this->patterns as $pattern) {
            if (preg_match($pattern['pattern'], $scan_content)) {
                $reasons[] = $pattern['reason'];
            }
        }
        
        // Cek string base64 panjang mencurigakan
        if (preg_match_all('/[A-Za-z0-9+\/=]{150,}/', $scan_content, $matches)) {
            foreach ($matches[0] as $long_string) {
                $decoded = @base64_decode($long_string, true);
                if ($decoded !== false && strlen($decoded) > 50) {
                    if (preg_match('/(eval|system|exec|shell_exec)/i', $decoded)) {
                        $reasons[] = 'Long base64 string terindikasi payload berbahaya';
                        break;
                    }
                }
            }
        }
        
        $reasons = array_unique($reasons);
        $this->cache[$cache_key] = $reasons;
        return $reasons;
    }
}

    if ($_GET['action'] !== 'login' && empty($_SESSION['logged_in'])) {

        case 'scan_init':
            $root = __DIR__;
            unset($_SESSION['scan_files'], $_SESSION['scan_results'], $_SESSION['scan_errors']);
            
            $files = fastFileScanner($root, $scan_extensions, $max_files_to_scan);
            $total = count($files);

                echo json_encode(['total' => 0, 'message' => 'Tidak ada file ditemukan']);
                exit;
            }
            
            $_SESSION['scan_files'] = $files;
            $_SESSION['scan_offset'] = 0;
            $_SESSION['scan_results'] = [];
            $_SESSION['scan_start_time'] = microtime(true);
            
            echo json_encode(['total' => $total, 'message' => "Ditemukan {$total} file untuk dipindai."]);
            exit;
        
        case 'scan_batch':
            if (!isset($_SESSION['scan_files'])) {
                http_response_code(400);
                echo json_encode(['error' => 'Scan belum diinisialisasi']);
                exit;
            }
            
            $offset = isset($_GET['offset']) ? max(0, intval($_GET['offset'])) : 0;
            $limit = $batch_limit;
            $files = $_SESSION['scan_files'];
            $total = count($files);
            
            $batch = array_slice($files, $offset, $limit);
            $findings = [];
            
            foreach ($batch as $filepath) {
                if (!is_file($filepath) || !is_readable($filepath)) continue;
                
                $size = @filesize($filepath);
                $content = @file_get_contents($filepath);
                if (!$content) continue;
                
                $relative = ltrim(str_replace(__DIR__, '', $filepath), '/\\');
                $reasons = $detector->detect($content, $relative);
                
                if (!empty($reasons)) {
                    $finding = [
                        'file' => $relative,
                        'reasons' => $reasons,
                        'size' => $size,
                        'modified' => date('Y-m-d H:i:s', filemtime($filepath))
                    ];
                    $findings[] = $finding;
                    $_SESSION['scan_results'][] = $finding;
                }
                unset($content);
            }
            
            $new_offset = $offset + count($batch);
            $_SESSION['scan_offset'] = $new_offset;
            $done = ($new_offset >= $total);
            $percentage = ($total > 0) ? ($new_offset / $total) * 100 : 100;
            $elapsed = microtime(true) - $_SESSION['scan_start_time'];
            
            echo json_encode([
                'next_offset' => $new_offset,
                'done' => $done,
                'findings' => $findings,
                'percentage' => round($percentage, 2),
                'elapsed' => round($elapsed, 2),
                'findings_count' => count($_SESSION['scan_results'])
            ]);
            exit;
            
        case 'get_results':
            echo json_encode(['results' => $_SESSION['scan_results'] ?? []]);
            exit;

        case 'delete_file':
            if (empty($_GET['file'])) {
                http_response_code(400);
                die(json_encode(['error' => 'File tidak ditentukan']));
            }
            $rel_path = $_GET['file'];
            if (strpos($rel_path, '../') !== false || strpos($rel_path, '..\\') !== false) {
                http_response_code(403);
                die(json_encode(['error' => 'Akses ilegal']));
            }
            $abs_path = realpath(__DIR__ . '/' . $rel_path);
            if ($abs_path && strpos($abs_path, __DIR__) === 0 && @unlink($abs_path)) {
                if (isset($_SESSION['scan_results'])) {
                    $_SESSION['scan_results'] = array_values(array_filter($_SESSION['scan_results'], function($v) use ($rel_path) {
                        return $v['file'] !== $rel_path;
                    }));
                }
                echo json_encode(['success' => true]);
            } else {
                echo json_encode(['error' => 'Gagal menghapus file']);
            }
            exit;
            
        case 'delete_all':
            $results = $_SESSION['scan_results'] ?? [];
            $deleted = 0;
            foreach ($results as $r) {
                $abs_path = realpath(__DIR__ . '/' . $r['file']);
                if ($abs_path && strpos($abs_path, __DIR__) === 0 && @unlink($abs_path)) {
                    $deleted++;
                }
            }
            $_SESSION['scan_results'] = [];
            echo json_encode(['success' => true, 'message' => "Berhasil menghapus {$deleted} file berbahaya."]);
            exit;
    }
}
