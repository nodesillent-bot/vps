/**
 * Fast file scanner menggunakan multiple methods
 */
function fastFileScanner($root, $extensions, $max_files) {
    $files = [];
    $patterns = [];
    
    // Buat pattern untuk glob
    foreach ($extensions as $ext) {
        $patterns[] = $root . '/**/*.' . $ext;
    }
    
    // Method 1: Glob (tercepat untuk struktur normal)
    $glob_files = glob($root . '/**/*.{' . implode(',', $extensions) . '}', GLOB_BRACE);
    if ($glob_files !== false) {
        $files = array_merge($files, $glob_files);
    }
    
    // Method 2: Jika glob tidak optimal, gunakan iterator dengan batasan
    if (count($files) < $max_files && $max_files > 0) {
        try {
            $iterator = new RecursiveIteratorIterator(
                new RecursiveDirectoryIterator($root, 
                    RecursiveDirectoryIterator::SKIP_DOTS | 
                    RecursiveDirectoryIterator::FOLLOW_SYMLINKS
                ),
                RecursiveIteratorIterator::LEAVES_ONLY
            );
            
            $count = 0;
            foreach ($iterator as $file) {
                if ($file->isFile()) {
                    $ext = strtolower($file->getExtension());
                    if (in_array($ext, $extensions)) {
                        $files[] = $file->getPathname();
                        $count++;
                        if ($count >= $max_files) break;
                    }
                }
            }
        } catch (Exception $e) {
            // Silent fail, lanjut dengan glob results
        }
    }
    
    // Filter unique files
    $files = array_unique($files);
    
    // Filter by file size
    $filtered = [];
    foreach ($files as $file) {
        if (is_file($file) && @filesize($file) <= 1024 * 1024) { // Max 1MB
            $filtered[] = $file;
        }
    }
    
    return array_slice($filtered, 0, $max_files);
}

/**
 * Ultra-fast pattern detection dengan caching
 */
class ShellDetector {
    private $patterns = [];
    private $compiled_patterns = null;
    private $cache = [];
    
    public function __construct() {
        $this->loadPatterns();
    }
    
    private function loadPatterns() {
        $this->patterns = [
            'dangerous_functions' => [
                'pattern' => '/\b(base64_decode|gzinflate|str_rot13|gzuncompress|strrev|rawurldecode|convert_uudecode|urldecode)\s*\(/i',
                'reason' => 'Fungsi encoding/decoding berbahaya'
            ],
            'system_commands' => [
                'pattern' => '/\b(system|shell_exec|exec|passthru|popen|proc_open|pcntl_exec|expect_popen)\s*\(\s*\$_(GET|POST|REQUEST|COOKIE|SERVER)\[/i',
                'reason' => 'Eksekusi perintah sistem dari input user'
            ],
            'backtick_exec' => [
                'pattern' => '/`\s*\$_(GET|POST|REQUEST|COOKIE)/i',
                'reason' => 'Backtick execution dari input user'
            ],
            'eval_hex' => [
                'pattern' => '/\$\{\s*[\'"]\x47\x4C\x4F\x42\x41\x4C\x53[\'"]\s*\[/i',
                'reason' => 'Eval dengan string hex'
            ],
            'assert_user' => [
                'pattern' => '/\bassert\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Assert dengan input user'
            ],
            'variable_variable' => [
                'pattern' => '/\$_?(GET|POST|REQUEST|COOKIE)\s*\[\s*[\'"]?\s*[\'"]?\s*\]\s*\(\s*\$_?(GET|POST|REQUEST)/i',
                'reason' => 'Variable variable dengan input user'
            ],
            'file_manipulation' => [
                'pattern' => '/\b(move_uploaded_file|copy|rename)\s*\(.*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Manipulasi file dari input user'
            ],
            'include_input' => [
                'pattern' => '/\b(include|require)(?:_once)?\s*\(\s*\$_(GET|POST|REQUEST|COOKIE)/i',
                'reason' => 'Include file dari input user'
            ],
            'curl_exec' => [
                'pattern' => '/\bcurl_exec\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Curl execution dari input user'
            ],
            'file_operation' => [
                'pattern' => '/\b(fopen|file_get_contents)\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'File operation dari input user'
            ],
            'eval_encoding' => [
                'pattern' => '/eval\s*\(\s*(base64_decode|str_rot13|gzuncompress|gzinflate)/i',
                'reason' => 'Eval dengan encoding'
            ],
            'preg_replace_e' => [
                'pattern' => '/preg_replace\s*\(.*\/e/i',
                'reason' => 'Preg_replace dengan modifier e (RCE)'
            ],
            'create_function' => [
                'pattern' => '/create_function\s*\(/i',
                'reason' => 'Create_function (dynamic code execution)'
            ],
            'callback_input' => [
                'pattern' => '/\b(array_map|array_filter|usort|uasort|uksort)\s*\(.*\$(GET|POST|REQUEST)/i',
                'reason' => 'Callback function dengan input user'
            ],
            'ob_start_callback' => [
                'pattern' => '/ob_start\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'OB Start dengan callback user'
            ],
            'extract_input' => [
                'pattern' => '/extract\s*\(\s*\$_(GET|POST|REQUEST)/i',
                'reason' => 'Extract dari input user'
            ]
        ];
    }
    
    public function detect($content, $filename = '') {
        $reasons = [];
        
        // Cek cache
        $cache_key = md5(substr($content, 0, 10000));
        if (isset($this->cache[$cache_key])) {
            return $this->cache[$cache_key];
        }
        
        // Scan hanya bagian penting (first 200KB)
        $scan_content = substr($content, 0, 204800);
        
        // Cek webshell terkenal (case insensitive)
        $webshell_keywords = ['c99shell', 'r57shell', 'b374k', 'wso', 'indoxploit', 
                              'crystal shell', 'ak-74', 'shell_finder', 'shell_runner'];
        foreach ($webshell_keywords as $keyword) {
            if (stripos($scan_content, $keyword) !== false) {
                $reasons[] = 'Web shell terkenal: ' . $keyword;
                break;
            }
        }
        
        // Cek semua pattern
        foreach ($this->patterns as $pattern) {
            if (preg_match($pattern['pattern'], $scan_content)) {
                $reasons[] = $pattern['reason'];
            }
        }
        
        // Deteksi base64 encoded strings yang panjang (indikasi shell)
        if (preg_match_all('/[A-Za-z0-9+\/=]{100,}/', $scan_content, $matches)) {
            foreach ($matches[0] as $long_string) {
                $decoded = @base64_decode($long_string, true);
                if ($decoded !== false && strlen($decoded) > 50) {
                    if (preg_match('/(eval|system|exec|shell_exec|base64_decode)/i', $decoded)) {
                        $reasons[] = 'Long base64 string mengandung kode berbahaya';
                        break;
                    }
                }
            }
        }
        
        $reasons = array_unique($reasons);
        $this->cache[$cache_key] = $reasons;
        
        // Limit cache size
        if (count($this->cache) > 1000) {
            array_shift($this->cache);
        }
        
        return $reasons;
    }
}

// ==================== AJAX HANDLERS ====================

if (isset($_GET['action'])) {
    // Authentication check
    if ($_GET['action'] !== 'login' && empty($_SESSION['logged_in'])) {
        http_response_code(403);
        die(json_encode(['error' => 'Unauthorized']));
    }
    
    header('Content-Type: application/json');
    
    switch ($_GET['action']) {
        
        case 'login':
            if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['password'])) {
                if ($_POST['password'] === $CORRECT_PASSWORD) {
                    session_regenerate_id(true);
                    $_SESSION['logged_in'] = true;
                    $_SESSION['login_time'] = time();
                    echo json_encode(['success' => true]);
                    exit;
                } else {
                    http_response_code(401);
                    echo json_encode(['error' => 'Password salah!']);
                    exit;
                }
            }
            break;
        
        case 'logout':
            $_SESSION = [];
            session_destroy();
            echo json_encode(['success' => true]);
            exit;
        
        case 'scan_init':
            $root = __DIR__;
            
            // Clear old session data
            unset($_SESSION['scan_files']);
            unset($_SESSION['scan_results']);
            unset($_SESSION['scan_errors']);
            
            // Start scanning
            $files = fastFileScanner($root, $scan_extensions, $max_files_to_scan);
            $total = count($files);
            
            if ($total === 0) {
                echo json_encode(['total' => 0, 'message' => 'Tidak ada file yang ditemukan']);
                exit;
            }
            
            // Store in session with chunking
            $_SESSION['scan_files'] = $files;
            $_SESSION['scan_offset'] = 0;
            $_SESSION['scan_results'] = [];
            $_SESSION['scan_errors'] = [];
            $_SESSION['scan_start_time'] = microtime(true);
            
            echo json_encode([
                'total' => $total,
                'message' => "Found {$total} files to scan"
            ]);
            exit;
        
        case 'scan_batch':
            if (!isset($_SESSION['scan_files'])) {
                http_response_code(400);
                echo json_encode(['error' => 'Scan belum diinisialisasi']);
                exit;
            }
            
            // Set timeout for this batch
            set_time_limit($timeout_per_batch);
            
            $offset = isset($_GET['offset']) ? max(0, intval($_GET['offset'])) : 0;
            $limit = isset($_GET['limit']) ? min(100, max(1, intval($_GET['limit']))) : $batch_limit;
            $files = $_SESSION['scan_files'];
            $total = count($files);
            
            // Get batch
            $batch = array_slice($files, $offset, $limit);
            $findings = [];
            $errors = [];
            
            foreach ($batch as $index => $filepath) {
                // Check if file exists and readable
                if (!is_file($filepath) || !is_readable($filepath)) {
                    continue;
                }
                
                // Check file size
                $size = @filesize($filepath);
                if ($size === false || $size > $max_filesize) {
                    continue;
                }
                
                // Read file efficiently
                $content = null;
                try {
                    if ($size < 102400) { // Small file: read all
                        $content = @file_get_contents($filepath);
                    } else { // Large file: read chunks
                        $handle = @fopen($filepath, 'rb');
                        if ($handle) {
                            $content = @fread($handle, $max_filesize);
                            @fclose($handle);
                        }
                    }
                } catch (Exception $e) {
                    $errors[] = ['file' => $filepath, 'error' => $e->getMessage()];
                    continue;
                }
                
                if ($content === false || empty($content)) {
                    continue;
                }
                
                // Detect malicious patterns
                $relative = ltrim(str_replace(__DIR__, '', $filepath), '/\\');
                $reasons = $detector->detect($content, $relative);
                
                if (!empty($reasons)) {
                    $finding = [
                        'file' => $relative,
                        'reasons' => $reasons,
                        'size' => $size,
                        'modified' => date('Y-m-d H:i:s', filemtime($filepath))
                    ];
                    $findings[] = $finding;
                    $_SESSION['scan_results'][] = $finding;
                }
                
                // Free memory
                unset($content);
            }
            
            $new_offset = $offset + count($batch);
            $_SESSION['scan_offset'] = $new_offset;
            $done = ($new_offset >= $total);
            
            // Calculate progress percentage
            $percentage = ($new_offset / $total) * 100;
            $elapsed = microtime(true) - $_SESSION['scan_start_time'];
            $estimated_total = $done ? $elapsed : ($elapsed / $new_offset) * $total;
            
            echo json_encode([
                'next_offset' => $new_offset,
                'done' => $done,
                'findings' => $findings,
                'errors' => $errors,
                'total_processed' => $new_offset,
                'total_files' => $total,
                'percentage' => round($percentage, 2),
                'elapsed' => round($elapsed, 2),
                'estimated_remaining' => $done ? 0 : round($estimated_total - $elapsed, 2),
                'findings_count' => count($_SESSION['scan_results'])
            ]);
            exit;
        
        case 'get_results':
            $results = isset($_SESSION['scan_results']) ? $_SESSION['scan_results'] : [];
            echo json_encode([
                'results' => $results,
                'total_findings' => count($results),
                'scan_completed' => isset($_SESSION['scan_offset']) && 
                                   isset($_SESSION['scan_files']) && 
                                   $_SESSION['scan_offset'] >= count($_SESSION['scan_files'])
            ]);
            exit;
        
        case 'delete_file':
            if (empty($_GET['file'])) {
                http_response_code(400);
                echo json_encode(['error' => 'File tidak disebutkan']);
                exit;
            }
            
            $rel_path = $_GET['file'];
            
            // Security: prevent directory traversal
            if (strpos($rel_path, '../') !== false || strpos($rel_path, '..\\') !== false) {
                http_response_code(403);
                echo json_encode(['error' => 'Path tidak valid']);
                exit;
            }
            
            $abs_path = realpath(__DIR__ . '/' . $rel_path);
            if (!$abs_path || strpos($abs_path, __DIR__ . DIRECTORY_SEPARATOR) !== 0) {
                http_response_code(403);
                echo json_encode(['error' => 'File di luar root direktori']);
                exit;
            }
            
            if (!file_exists($abs_path)) {
                http_response_code(404);
                echo json_encode(['error' => 'File tidak ditemukan']);
                exit;
            }
            
            // Try to delete
            if (@unlink($abs_path)) {
                // Remove from results
                if (isset($_SESSION['scan_results'])) {
                    $_SESSION['scan_results'] = array_values(array_filter($_SESSION['scan_results'], function($item) use ($rel_path) {
                        return $item['file'] !== $rel_path;
                    }));
                }
                echo json_encode(['success' => true, 'message' => 'File berhasil dihapus']);
            } else {
                http_response_code(500);
                echo json_encode(['error' => 'Gagal menghapus file (permission denied)']);
            }
            exit;
        
        case 'delete_all':
            if (!isset($_SESSION['scan_results']) || empty($_SESSION['scan_results'])) {
                echo json_encode(['success' => false, 'message' => 'Tidak ada file untuk dihapus']);
                exit;
            }
            
            $deleted = 0;
            $failed = 0;
            
            foreach ($_SESSION['scan_results'] as $result) {
                $abs_path = realpath(__DIR__ . '/' . $result['file']);
                if ($abs_path && strpos($abs_path, __DIR__) === 0 && @unlink($abs_path)) {
                    $deleted++;
                } else {
                    $failed++;
                }
            }
            
            $_SESSION['scan_results'] = [];
            
            echo json_encode([
                'success' => true,
                'deleted' => $deleted,
                'failed' => $failed,
                'message' => "Deleted {$deleted} files, failed {$failed}"
            ]);
            exit;
        
        case 'export_results':
            $results = isset($_SESSION['scan_results']) ? $_SESSION['scan_results'] : [];
            $export = [
                'timestamp' => date('Y-m-d H:i:s'),
                'scanned_files' => isset($_SESSION['scan_offset']) ? $_SESSION['scan_offset'] : 0,
                'total_findings' => count($results),
                'findings' => $results
            ];
            
            header('Content-Type: application/json');
            header('Content-Disposition: attachment; filename="scan_results_' . date('Ymd_His') . '.json"');
            echo json_encode($export, JSON_PRETTY_PRINT);
            exit;
        
        case 'server_status':
            $status = [
                'php_version' => PHP_VERSION,
                'memory_usage' => round(memory_get_usage(true) / 1024 / 1024, 2) . ' MB',
                'memory_limit' => ini_get('memory_limit'),
                'max_execution_time' => ini_get('max_execution_time'),
                'upload_max_filesize' => ini_get('upload_max_filesize'),
                'post_max_size' => ini_get('post_max_size'),
                'server_software' => $_SERVER['SERVER_SOFTWARE'] ?? 'Unknown',
                'safe_mode' => ini_get('safe_mode') ? 'On' : 'Off',
                'disabled_functions' => ini_get('disable_functions')
            ];
            echo json_encode($status);
            exit;
        
        default:
            http_response_code(400);
            echo json_encode(['error' => 'Action tidak dikenal']);
            exit;
    }
}

<script>
// State management

// DOM elements

const resultsContainer = document.getElementById('resultsContainer');
const totalFilesEl = document.getElementById('totalFiles');
const findingsCountEl = document.getElementById('findingsCount');
const progressPercentEl = document.getElementById('progressPercent');
const timeElapsedEl = document.getElementById('timeElapsed');

// Helper functions
function formatTime(seconds) {
    if (seconds < 60) return Math.round(seconds) + 's';
    const mins = Math.floor(seconds / 60);
    const secs = Math.round(seconds % 60);
    return mins + 'm ' + secs + 's';
}

function updateStats(processed, total, findingsTotal, percentage) {
    totalFilesEl.textContent = processed + ' / ' + total;
    findingsCountEl.textContent = findingsTotal;
    progressPercentEl.textContent = percentage + '%';
    
    if (scanStartTime) {
        const elapsed = (Date.now() - scanStartTime) / 1000;
        timeElapsedEl.textContent = formatTime(elapsed);
    }
}

function renderFindings() {
    if (findings.length === 0) {
        resultsContainer.innerHTML = '<div class="empty-state"><p>✅ No suspicious files found! Your server is clean.</p></div>';
        return;
    }
    
    let html = '<table><thead><tr><th>File Path</th><th>Reasons</th><th>Size</th><th>Modified</th><th>Action</th></tr></thead><tbody>';
    
    findings.forEach((finding, index) => {
        const reasonsHtml = finding.reasons.map(r => `<span>${escapeHtml(r)}</span>`).join('');
        html += `
            <tr>
                <td class="file-path">${escapeHtml(finding.file)}</td>
                <td class="reasons">${reasonsHtml}</td>
                <td>${formatBytes(finding.size || 0)}</td>
                <td>${finding.modified || '-'}</td>
                <td><button class="delete-btn" onclick="deleteFile('${escapeHtml(finding.file).replace(/'/g, "\\'")}')">Delete</button></td>
            </tr>
        `;
    });
    
    html += '</tbody></table>';
    resultsContainer.innerHTML = html;
}

function formatBytes(bytes) {
    if (bytes === 0) return '0 B';
    const k = 1024;
    const sizes = ['B', 'KB', 'MB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
}

function escapeHtml(str) {
    return str.replace(/[&<>]/g, function(m) {
        if (m === '&') return '&amp;';
        if (m === '<') return '&lt;';
        if (m === '>') return '&gt;';
        return m;
    });
}

async function deleteFile(file) {
    if (!confirm('⚠️ WARNING: Are you sure you want to delete:\n\n' + file + '\n\nThis action cannot be undone!')) return;
    
    try {
        const response = await fetch(`?action=delete_file&file=${encodeURIComponent(file)}`);
        const data = await response.json();
        
        if (data.success) {
            // Remove from findings array
            findings = findings.filter(f => f.file !== file);
            renderFindings();
            findingsCountEl.textContent = findings.length;
            
            // Show success message
            showNotification('File deleted successfully', 'success');
        } else {
            alert('Failed to delete: ' + (data.error || 'Unknown error'));
        }
    } catch (error) {
        alert('Error: ' + error.message);
    }
}

    const totalFindings = findings.length;
    if (totalFindings === 0) {
        alert('No files to delete');
        return;
    }
    
    if (!confirm(`⚠️⚠️⚠️ DANGER ZONE ⚠️⚠️⚠️\n\nYou are about to delete ALL ${totalFindings} suspicious files.\n\nThis action is IRREVERSIBLE!\n\nType "DELETE ALL" to confirm:`)) {

    }

async function startScan() {
    if (isScanning) {
        alert('Scan already in progress!');
        return;
    }
    
    isScanning = true;
    startScanBtn.disabled = true;
    startScanBtn.textContent = 'Scanning...';
    progressSection.style.display = 'block';
    findings = [];
    currentOffset = 0;
    scanStartTime = Date.now();
    
    // Hide export and delete buttons initially
    exportResultsBtn.style.display = 'none';
    deleteAllBtn.style.display = 'none';
    
    try {
        // Initialize scan
        const initResponse = await fetch('?action=scan_init');
        const initData = await initResponse.json();
        
        if (initData.error) throw new Error(initData.error);
        
        totalFiles = initData.total;
        
        if (totalFiles === 0) {
            progressText.textContent = 'No files to scan';
            progressBar.style.width = '100%';
            progressBar.textContent = '100%';
            resultsContainer.innerHTML = '<div class="empty-state"><p>📁 No files found to scan</p></div>';
            isScanning = false;
            startScanBtn.disabled = false;
            startScanBtn.textContent = 'Start Full Scan';
            return;
        }
        
        progressText.textContent = `Found ${totalFiles} files to scan`;
        
        // Process batches
        while (currentOffset < totalFiles && isScanning) {
            const batchResponse = await fetch(`?action=scan_batch&offset=${currentOffset}&limit=50`);
            const batchData = await batchResponse.json();
            
            if (batchData.error) throw new Error(batchData.error);
            
            // Add new findings
            if (batchData.findings && batchData.findings.length > 0) {
                findings.push(...batchData.findings);
                renderFindings();
            }
            
            currentOffset = batchData.next_offset;
            const percentage = batchData.percentage || ((currentOffset / totalFiles) * 100);
            const progressPercent = Math.round(percentage);
            
            progressBar.style.width = progressPercent + '%';
            progressBar.textContent = progressPercent + '%';
            progressText.textContent = `Scanning... ${currentOffset} of ${totalFiles} files (${findings.length} suspicious found)`;
            
            updateStats(currentOffset, totalFiles, findings.length, progressPercent);
            
            if (batchData.done) break;
            
            // Small delay to prevent overwhelming the server
            await new Promise(resolve => setTimeout(resolve, 50));
        }
        
        // Scan completed
        progressText.textContent = `✅ Scan completed! Found ${findings.length} suspicious files.`;
        progressBar.style.backgroundColor = '#48bb78';
        
        if (findings.length > 0) {
            exportResultsBtn.style.display = 'inline-block';
            deleteAllBtn.style.display = 'inline-block';
        }
        
    } catch (error) {
        console.error('Scan error:', error);
        progressText.textContent = '❌ Scan failed: ' + error.message;
        resultsContainer.innerHTML = `<div class="empty-state"><p>❌ Error: ${escapeHtml(error.message)}</p></div>`;
    } finally {
        isScanning = false;
        startScanBtn.disabled = false;
        startScanBtn.textContent = 'Start Full Scan';
    }
}

        console.error('Refresh error:', error);
    }
