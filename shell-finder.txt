$scan_extensions = ['php','phtml','php5','php7','phps','inc','txt','htaccess','html','htm','shtml','cgi','pl','py','sh'];
$max_filesize = 512 * 1024;
$batch_limit = 40;

// Fungsi deteksi pattern berbahaya dengan PHP murni
function detectMaliciousPatterns($content) {
    $reasons = [];
    
    // Deteksi base64_decode, gzinflate, str_rot13, gzuncompress, strrev, rawurldecode, convert_uudecode, urldecode
    if (preg_match('/\b(base64_decode|gzinflate|str_rot13|gzuncompress|strrev|rawurldecode|convert_uu解码|urldecode)\s*\(/i', $content)) {
        $reasons[] = 'Fungsi encoding/decoding berbahaya';
    }
    
    // Deteksi system, shell_exec, exec, passthru, popen, proc_open, pCntl_exec, expect_popen
    if (preg_match('/\b(system|shell_exec|exec|passthru|popen|proc_open|pcntl_exec|expect_popen)\s*\(\s*\$_(GET|POST|REQUEST|COOKIE|SERVER)\[/i', $content)) {
        $reasons[] = 'Eksekusi perintah sistem dari input user';
    }
    
    // Deteksi backtick execution dengan $_GET/$_POST
    if (preg_match('/`\s*\$_(GET|POST|REQUEST|COOKIE)/i', $content)) {
        $reasons[] = 'Backtick execution dari input user';
    }
    
    // Deteksi web shell terkenal
    $webshell_patterns = [
        'c99shell', 'r57shell', 'b374k', 'wso shell', 'indoxploit', 
        'php shell', 'crystal shell', 'ak-74', 'simple shell', 
        'evil shell', 'web shell'
    ];
    foreach ($webshell_patterns as $shell) {
        if (preg_match('/\b' . preg_quote($shell, '/') . '\b/i', $content)) {
            $reasons[] = 'Web shell terkenal terdeteksi';
            break;
        }
    }
    
    // Deteksi eval dengan string hex
    if (preg_match('/\$\{\s*[\'"]\x47\x4C\x4F\x42\x41\x4C\x53[\'"]\s*\[/i', $content)) {
        $reasons[] = 'Eval dengan string hex';
    }
    
    // Deteksi assert dengan $_GET/$_POST/$_REQUEST
    if (preg_match('/\bassert\s*\(\s*\$_(GET|POST|REQUEST)/i', $content)) {
        $reasons[] = 'Assert dengan input user';
    }
    
    // Deteksi $_GET/$_POST/$_REQUEST dengan kurung dan eval
    if (preg_match('/\$_?(GET|POST|REQUEST|COOKIE)\s*\[\s*[\'"]?\s*[\'"]?\s*\]\s*\(\s*\$_?(GET|POST|REQUEST)/i', $content)) {
        $reasons[] = 'Variable variable dengan input user';
    }
    
    // Deteksi move_uploaded_file, copy, rename dengan $_GET/$_POST/$_REQUEST
    if (preg_match('/\b(move_uploaded_file|copy|rename)\s*\(.*\$_(GET|POST|REQUEST)/i', $content)) {
        $reasons[] = 'Manipulasi file dari input user';
    }
    
    // Deteksi include/require dengan $_GET/$_POST/$_REQUEST
    if (preg_match('/\b(include|require)(?:_once)?\s*\(\s*\$_(GET|POST|REQUEST|COOKIE)/i', $content)) {
        $reasons[] = 'Include file dari input user';
    }
    
    // Deteksi curl_exec dengan $_GET/$_POST/$_REQUEST
    if (preg_match('/\bcurl_exec\s*\(\s*\$_(GET|POST|REQUEST)/i', $content)) {
        $reasons[] = 'Curl execution dari input user';
    }
    
    // Deteksi fopen, file_get_contents dengan $_GET/$_POST/$_REQUEST
    if (preg_match('/\b(fopen|file_get_contents)\s*\(\s*\$_(GET|POST|REQUEST)/i', $content)) {
        $reasons[] = 'File operation dari input user';
    }
    
    // Deteksi eval dengan encoding
    if (preg_match('/eval\s*\(\s*(base64_decode|str_rot13|gzuncompress|gzinflate)/i', $content)) {
        $reasons[] = 'Eval dengan encoding';
    }
    
    // Deteksi preg_replace dengan modifier e (deprecated tapi masih berbahaya)
    if (preg_match('/preg_replace\s*\(.*\/e/i', $content)) {
        $reasons[] = 'Preg_replace dengan modifier e (remote code execution)';
    }
    
    // Deteksi create_function
    if (preg_match('/create_function\s*\(/i', $content)) {
        $reasons[] = 'Create_function (dynamic code execution)';
    }
    
    // Deteksi callback berbahaya
    if (preg_match('/\b(array_map|array_filter|usort|uasort|uksort)\s*\(.*\$(GET|POST|REQUEST)/i', $content)) {
        $reasons[] = 'Callback function dengan input user';
    }
    
    return $reasons;
}

if (isset($_GET['action'])) {
    if ($_GET['action'] !== 'login' && empty($_SESSION['logged_in'])) {
        http_response_code(403);
        die(json_encode(['error' => 'Unauthorized']));
    }

    switch ($_GET['action']) {

        case 'login':
            if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['password'])) {
                if ($_POST['password'] === $CORRECT_PASSWORD) {
                    session_regenerate_id(true);
                    $_SESSION['logged_in'] = true;
                    header('Location: ' . $_SERVER['PHP_SELF']);
                    exit;
                } else {
                    $login_error = 'Password salah!';
                }
            }
            break;

        case 'logout':
            $_SESSION = [];
            session_destroy();
            header('Location: ' . $_SERVER['PHP_SELF']);
            exit;

        case 'scan_init':
            $root = __DIR__;
            $files = [];
            try {
                $iterator = new RecursiveIteratorIterator(
                    new RecursiveDirectoryIterator($root, RecursiveDirectoryIterator::SKIP_DOTS)
                );
                foreach ($iterator as $item) {
                    if ($item->isFile()) {
                        $ext = strtolower(pathinfo($item->getFilename(), PATHINFO_EXTENSION));
                        if (in_array($ext, $scan_extensions)) {
                            $files[] = $item->getPathname();
                        }
                    }
                }
            } catch (Exception $e) {
                http_response_code(500);
                die(json_encode(['error' => 'Gagal scan direktori: ' . $e->getMessage()]));
            }
            $_SESSION['scan_files'] = $files;
            $_SESSION['scan_offset'] = 0;
            $_SESSION['scan_results'] = [];
            echo json_encode(['total' => count($files)]);
            exit;

        case 'scan_batch':
            if (!isset($_SESSION['scan_files'])) {
                http_response_code(400);
                die(json_encode(['error' => 'Scan belum diinisialisasi']));
            }
            $offset = isset($_GET['offset']) ? max(0, intval($_GET['offset'])) : 0;
            $limit = isset($_GET['limit']) ? min(100, max(1, intval($_GET['limit']))) : $batch_limit;
            $files = $_SESSION['scan_files'];
            $total = count($files);
            $batch = array_slice($files, $offset, $limit);
            $findings = [];

            foreach ($batch as $filepath) {
                if (!file_exists($filepath) || filesize($filepath) > $max_filesize) {
                    continue;
                }
                $content = @file_get_contents($filepath);
                if ($content === false) continue;

                // Gunakan fungsi deteksi pattern yang sudah dibuat
                $reasons = detectMaliciousPatterns($content);
                
                if (!empty($reasons)) {
                    $relative = ltrim(str_replace(__DIR__, '', $filepath), '/\\');
                    $findings[] = ['file' => $relative, 'reasons' => $reasons];
                    $_SESSION['scan_results'][] = ['file' => $relative, 'reasons' => $reasons];
                }
            }

            $new_offset = $offset + count($batch);
            $_SESSION['scan_offset'] = $new_offset;
            $done = ($new_offset >= $total);

            echo json_encode([
                'next_offset'     => $new_offset,
                'done'            => $done,
                'findings'        => $findings,
                'total_processed' => $new_offset,
                'total_files'     => $total
            ]);
            exit;

        case 'delete':
            if (empty($_GET['file'])) {
                http_response_code(400);
                die(json_encode(['error' => 'File tidak disebutkan']));
            }
            $rel_path = $_GET['file'];
            if (strpos($rel_path, '../') !== false || strpos($rel_path, '..\\') !== false) {
                http_response_code(403);
                die(json_encode(['error' => 'Path tidak valid']));
            }
            $abs_path = realpath(__DIR__ . '/' . $rel_path);
            if (!$abs_path || strpos($abs_path, __DIR__ . DIRECTORY_SEPARATOR) !== 0) {
                http_response_code(403);
                die(json_encode(['error' => 'File di luar root direktori']));
            }
            if (!file_exists($abs_path)) {
                http_response_code(404);
                die(json_encode(['error' => 'File tidak ditemukan']));
            }
            if (@unlink($abs_path)) {
                if (isset($_SESSION['scan_results'])) {
                    $_SESSION['scan_results'] = array_filter($_SESSION['scan_results'], function($item) use ($rel_path) {
                        return $item['file'] !== $rel_path;
                    });
                }
                echo json_encode(['success' => true]);
            } else {
                http_response_code(500);
                echo json_encode(['error' => 'Gagal menghapus file (izin?)']);
            }
            exit;

        default:
            http_response_code(400);
            die(json_encode(['error' => 'Action tidak dikenal']));
    }
}

function updateProgress(current, total) {

async function deleteFile(file) {
    if (!confirm('Yakin hapus:\n' + file + '\nTindakan ini tidak bisa dibatalkan!')) return;
    try {
        let resp = await fetch(`?action=delete&file=${encodeURIComponent(file)}`);
        let data = await resp.json();
        if (data.success) {
            let rows = findingsBody.querySelectorAll('tr');
            rows.forEach(row => { if (row.cells[0].textContent === file) row.remove(); });
            if (findingsBody.rows.length === 0) {
                findingsTable.style.display = 'none';
                emptyMessage.style.display = '';
            }
        } else {
            alert('Gagal: ' + (data.error || 'Error'));
        }
    } catch(e) { alert('Error: ' + e.message); }
}

    findingsBody.innerHTML = '';
